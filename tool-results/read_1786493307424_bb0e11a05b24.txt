    26→function formatDate(ts: number): string {
    27→  return new Date(ts).toLocaleDateString('pt-BR', {
    28→    day: '2-digit',
    29→    month: '2-digit',
    30→    year: 'numeric',
    31→    hour: '2-digit',
    32→    minute: '2-digit',
    33→  })
    34→}
    35→
    36→function timeAgo(ts: number): string {
    37→  const diff = Date.now() - ts
    38→  const minutes = Math.floor(diff / 60000)
    39→  if (minutes < 1) return 'agora'
    40→  if (minutes < 60) return `${minutes}min atrás`
    41→  const hours = Math.floor(minutes / 60)
    42→  if (hours < 24) return `${hours}h atrás`
    43→  const days = Math.floor(hours / 24)
    44→  return `${days}d atrás`
    45→}
